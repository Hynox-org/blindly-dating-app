import os
import json
import logging
import psycopg2
import psycopg2.extras

logger = logging.getLogger()
logger.setLevel(logging.INFO)

def get_db_connection():
    database_url = os.environ.get('DATABASE_URL')
    return psycopg2.connect(database_url, cursor_factory=psycopg2.extras.RealDictCursor)

def lambda_handler(event, context):
    try:
        body = json.loads(event.get('body', '{}'))
        profile_id = body.get('profile_id')
        
        if not profile_id:
            return { 'statusCode': 400, 'body': json.dumps({'error': 'profile_id is required'}) }

        conn = get_db_connection()
        cursor = conn.cursor()

        # 1. Look up the caller's profile and active mode filters inside a JOIN
        cursor.execute("""
            SELECT 
                p.id,
                p.user_id,
                p.gender,
                ST_AsText(p.location_geom) as loc_text, 
                ST_AsText(p.passport_location_geom) as pass_text,
                p.current_mode,
                pm.filters
            FROM profiles p
            JOIN profile_modes pm ON p.id = pm.profile_id AND pm.mode = p.current_mode
            WHERE (p.id = %s OR p.user_id = %s) AND pm.is_active = True
            LIMIT 1
        """, (profile_id, profile_id))
        
        me = cursor.fetchone()
        
        if not me:
            return { 'statusCode': 404, 'body': json.dumps({'error': 'Profile or Active Mode not found'}) }

        my_geom = me.get('pass_text') or me.get('loc_text')
        
        if not my_geom:
            return { 'statusCode': 400, 'body': json.dumps({'error': 'Caller has no location set'}) }

        mode = me['current_mode'] or 'date'
        my_gender = me.get('gender')
        raw_filters = me.get('filters') or {}

        # 2. Extract Filters dynamically from the Flutter App State
        gender_pref = raw_filters.get('genderPreference', 'Everyone')
        min_age = raw_filters.get('minAge', 18)
        max_age = raw_filters.get('maxAge', 100)
        dist_limit_km = raw_filters.get('distanceLimit', 200) # Defaults to 200km sweep
        
        # 3. Determine Gender Match Logic
        target_genders = []
        if gender_pref == 'Women':
            target_genders = ['F']
        elif gender_pref == 'Men':
            target_genders = ['M']
        else: 
            # If set to 'Everyone' or missing, fallback to standard dating logic
            if my_gender in ['NB', 'Prefer Not', None]:
                 target_genders = ['M', 'F', 'NB', 'Prefer Not']
            elif mode == 'date':
                 # Heterosexual logic for Date mode
                 if my_gender == 'M': target_genders = ['F']
                 elif my_gender == 'F': target_genders = ['M']
            elif mode == 'bff':
                 # Same-sex logic for BFF mode
                 if my_gender == 'M': target_genders = ['M']
                 elif my_gender == 'F': target_genders = ['F']

        if not target_genders:
            target_genders = ['M', 'F', 'NB', 'Prefer Not']

        # 4. Execute Highly Explicit Geographic Dating Query
        cursor.execute("""
            SELECT p.id 
            FROM profiles p
            WHERE p.id != %s 
              AND p.current_mode = %s
              AND p.gender::text = ANY(%s::text[])
              AND (p.location_geom IS NOT NULL OR p.passport_location_geom IS NOT NULL)
              /* Filter by Age Boundary */
              AND EXTRACT(YEAR FROM age(current_date, p.birth_date)) BETWEEN %s AND %s
              /* Filter mathematically by Radius in meters */
              AND ST_DWithin(
                  COALESCE(p.passport_location_geom, p.location_geom), 
                  (SELECT COALESCE(passport_location_geom, location_geom) FROM profiles WHERE id = %s), 
                  %s
              )
              /* Strict exclusion of history (4 Swipe Scenarios) */
              AND NOT EXISTS (
                  SELECT 1 FROM swipes s 
                  WHERE 
                      (s.actor_id = %s AND s.target_id = p.id)
                      OR 
                      (s.actor_id = p.id AND s.target_id = %s AND s.action_type IN ('like', 'super_like'))
              )
            ORDER BY COALESCE(p.passport_location_geom, p.location_geom) <-> (SELECT COALESCE(passport_location_geom, location_geom) FROM profiles WHERE id = %s) ASC
            LIMIT 20
        """, (
            me['id'],                   # p.id != me 
            mode,                       # Match exactly the same mode
            target_genders,             # Enforce gender matching boundaries
            min_age, max_age,           # Extracted App Age Ranges
            me['id'],                   # User coordinates for radius sweep
            dist_limit_km * 1000,       # Distance limits in meters
            me['id'],                   # Scenario 1: I swiped on them
            me['id'],                   # Scenario 2: They 'liked'/'super_liked' me
            me['id']                    # Anchor point for ASC distance ordering
        ))
        
        profile_ids = [row['id'] for row in cursor.fetchall()]

        return {
            'statusCode': 200,
            'body': json.dumps({'profiles': profile_ids, 'count': len(profile_ids), 'exhausted': len(profile_ids) < 20})
        }

    except Exception as e:
        return { 'statusCode': 500, 'body': json.dumps({'error': str(e)}) }
    finally:
        if 'conn' in locals(): conn.close()
